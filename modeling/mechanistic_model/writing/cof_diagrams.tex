  % ─── Foldable branch ─────────────────────────────────────────────────
  \node[branchpanel={foldGreen}{foldGreenBg},
        minimum width=4.2cm, minimum height=2.8cm]

  % Fraction label: f_eff = f_base + f_gain·(n-1)

  % ─── Non-foldable branch ─────────────────────────────────────────────

  \node[font=\small, text=black, anchor=north]
